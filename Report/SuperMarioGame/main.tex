% Inline code that is allowed to wrap.
%
% The report is full of paths and identifiers - docs/issues/code_audit_...md,
% PlayingState::flushPendingSpawns - and plain \texttt refuses to break them,
% pushing lines well into the margin. This permits a break after the characters
% such names are made of, without hyphenating anything.
\newcommand{\codebreak}[1]{\texttt{\seqsplitcode{#1}}}
\makeatletter
\newcommand{\seqsplitcode}[1]{%
  \begingroup
  \def\-{\discretionary{}{}{}}%
  \catcode`\/=13 \catcode`\_=13 \catcode`\.=13 \catcode`\:=13 \catcode`\-=13
  \scantokens{#1\noexpand}%
  \endgroup}
\makeatother
\begingroup
\catcode`\/=13 \catcode`\_=13 \catcode`\.=13 \catcode`\:=13 \catcode`\-=13
\gdef/{\char`\/\allowbreak}
\gdef_{\char`\_\allowbreak}
\gdef.{\char`\.\allowbreak}
\gdef:{\char`\:\allowbreak}
\gdef-{\char`\-\allowbreak}
\endgroup

% Let the paragraph builder stretch inter-word spaces further as a last
% resort before giving up and reporting an Overfull \hbox (plan section
% 4.1's build.sh guard fails the build on any of those over 5pt). This is
% the standard, content-independent remedy for the ordinary tight line a
% long report accumulates - one \codebreak'd identifier landing a few points
% short of the margin - as opposed to \seqsplitcode's per-character breaks,
% which target a SPECIFIC token with nowhere at all to break. Both are
% needed: this pass would not, by itself, have fixed IMovementStrategy
% overflowing a whole table column.
\setlength{\emergencystretch}{8em}
