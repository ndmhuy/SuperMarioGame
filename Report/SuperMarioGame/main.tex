% Inline code that is allowed to wrap.
%
% The report is full of paths and identifiers - docs/issues/code_audit_...md,
% PlayingState::flushPendingSpawns - and plain \texttt refuses to break them,
% pushing lines well into the margin. This permits a break after the characters
% such names are made of, without hyphenating anything.
\newcommand{\codebreak}[1]{\texttt{\seqsplitcode{#1}}}
\makeatletter
\newcommand{\seqsplitcode}[1]{%
  \begingroup
  \def\-{\discretionary{}{}{}}%
  \catcode`\/=13 \catcode`\_=13 \catcode`\.=13 \catcode`\:=13 \catcode`\-=13
  \scantokens{#1\noexpand}%
  \endgroup}
\makeatother
\begingroup
\catcode`\/=13 \catcode`\_=13 \catcode`\.=13 \catcode`\:=13 \catcode`\-=13
\gdef/{\char`\/\allowbreak}
\gdef_{\char`\_\allowbreak}
\gdef.{\char`\.\allowbreak}
\gdef:{\char`\:\allowbreak}
\gdef-{\char`\-\allowbreak}
\endgroup
